# 图库常用语句

# 指标

## 原子指标

```postgresql
match {
    type: Indicator,
    as: ind,
    where: (code = 'M0001')
}.out('IndicatorIsDrilledByDimension'){
    as: dim
}.out('DimensionBoundToATT'){
    as: attr
}.in('LEContainATT'){
    as: le
} return ind,dim,attr,le
```

```postgresql
match {
    type: Indicator,
    as: ind,
    where: (code = 'M0001')
}.out('IndicatorIsDrilledByDimension'){
    as: dim,
    where: (code='D002')
}.out('DimensionReferencesATT'){
    as: attr
}.in('LEContainATT'){
    as: le
} return ind,dim,attr,le
```
```postgresql
match {
    type: Indicator,
    as: ind,
    where: (code = 'M0001')
}.out('IndicatorIsDrilledByDimension'){
    as: dim,
    where: (code='D005')
}.out('DimensionReferencesATT'){
    as: attr
}.in('ATTHasHierarchyParent'){
	while: ( $depth < 10 ),
    as: hAttr
}.in('LEContainATT'){
    as: le
} return ind,dim,attr,hAttr,le

```

## 复合指标

```postgresql
match {
    type: Indicator,
    as: cind,
    where: (code = 'M0002')
}.out('IndicatorCalculateToIndicator'){
    as: ind
}.out('IndicatorIsCalculatedFromATT','IndicatorIsDrilledByDimension'){
    as: attr
}.in('LEContainATT'){
    as: le
} return cind,ind,attr,le
```

# 逻辑实体

```postgresql
match {
    type: LogicalEntity,
    as: src,
    where: (code ='LE000006')
}.out('LEAssociateLE'){
    while: ( $depth < 10 ),
    type: LogicalEntity,
    as: tar,
    where: (code ='LE000007')
} return src,tar
```
```postgresql
MATCH (s:LogicalEntity)-[r:LEAssociateLE]->(t:LogicalEntity)
WHERE s.code IN ['LE000001','LE000002','LE000010','LE000011']
  AND t.code IN ['LE000001','LE000002','LE000010','LE000011']
RETURN s, r, t
```

# 维度

```postgresql
MATCH {
    type: Dimension,
    as: dim,
    where: (code = 'D005')
}.outE('DimensionReferencesATT'){
    as: eDimAttr
}.inV(){
    as: attr
}.in('ATTHasHierarchyParent'){
    while: ( $depth <= 10 ),
    as: hAttr
}.inE('LEContainATT'){
    as: eLeAttr
}.outV(){
    as: le
}
RETURN dim, eDimAttr, attr, hAttr, eLeAttr, le

```
